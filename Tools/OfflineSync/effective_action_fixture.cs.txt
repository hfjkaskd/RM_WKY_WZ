static class Checks {
    static int passed;
    static void Check(bool value, string name) { if(!value) throw new Exception(name); Console.WriteLine("PASS " + name); passed++; }
    static (Whitebox.Foundation.CoreGameFlow core, Whitebox.Foundation.OriginalMatchGroupExplosion explosion, Whitebox.Foundation.ExplosionServices services) Setup() {
        AccountModule.Instance = new(); SaveDataUtils.TeachData = new(); SaveDataUtils.Saves = 0; Teach01Module.Instance = new();
        FlowModule.AutomaticPopupFlowBlocked = false;
        var core = new Whitebox.Foundation.CoreGameFlow();
        Whitebox.Foundation.RoyalMatchSessionBridge.ActiveController = new(){ CoreFlow = core };
        var services = new Whitebox.Foundation.ExplosionServices { Core = core };
        var explosion = new Whitebox.Foundation.OriginalMatchGroupExplosion(new(), services);
        core.Bind(explosion); return(core, explosion, services);
    }
    static void Match(Whitebox.Foundation.OriginalMatchGroupExplosion explosion, Whitebox.Foundation.ExplosionServices services) {
        services.Cell = new(); var group = new Whitebox.Foundation.OriginalMatchGroupSnapshot(); group.Cells.Add(services.Cell);
        explosion.ExplodeMatchGroup(group, new());
    }
    static void Main() {
        var x = Setup();
        Match(x.explosion,x.services);
        Check(SaveDataUtils.Saves == 0, "Initial automatic matches with no accepted move do not mark first action");
        Check(SaveDataUtils.TeachData.newbieGiftRequest == NewbieGiftRequestState.NotTriggered, "Invalid input without a consumed move leaves gift state untouched");
        x.core.game.Moves.MadeMoves = 1; x.services.Remove = false; Match(x.explosion,x.services);
        Check(SaveDataUtils.Saves == 0, "A protected/non-removed match item does not report success");
        x.services.Remove = true; x.core.IsSynthesisTutorialActive = x.core.suppressTutorialSynthesis = true;
        Check(!x.core.SynthesisAllowed, "Existing tutorial synthesis reward filtering remains active");
        Match(x.explosion,x.services);
        Check(SaveDataUtils.TeachData.newbieGiftRequest == NewbieGiftRequestState.Background && SaveDataUtils.Saves == 1, "Tutorial match persists offline first action immediately");
        Check(!x.core.CanShowAutomaticPopup, "Tutorial still blocks gift presentation");
        Match(x.explosion,x.services); FlowModule.ReportFirstEffectiveAction();
        Check(SaveDataUtils.Saves == 1, "Further matches and legacy report calls cannot create another claim");

        x = Setup(); x.core.game.Moves.MadeMoves = 1; x.services.WinOnRemove = true; Match(x.explosion,x.services);
        Check(!x.core.game.State.IsPlaying() && SaveDataUtils.TeachData.newbieGiftRequest == NewbieGiftRequestState.Background, "First move winning during item removal still persists the claim");
        Check(!x.core.CanShowAutomaticPopup, "Winning board cannot open a gift over win presentation");

        x = Setup(); AccountModule.Instance.AccountReady = true; x.core.game.Moves.MadeMoves = 1; Match(x.explosion,x.services);
        Check(SaveDataUtils.TeachData.newbieGiftRequest == NewbieGiftRequestState.NextLogin, "Online effective action retains online claim semantics");
        x = Setup(); x.core.effectiveActionBaselineMoves = x.core.game.Moves.MadeMoves = 5; Match(x.explosion,x.services);
        Check(SaveDataUtils.Saves == 0, "Restored board auto matches cannot pretend to be a new player action");
        x.core.game.Moves.MadeMoves++; Match(x.explosion,x.services);
        Check(SaveDataUtils.Saves == 1, "First new move after board restore reports successfully");
        x = Setup(); x.core.game.Moves.MadeMoves = 1; x.explosion.BindEffectiveActionObserver(null,null); Match(x.explosion,x.services);
        Check(SaveDataUtils.Saves == 0, "Released observer does not retain or notify old bridge");

        x = Setup(); var teach = new Teach01UnityServices();
        Check(teach.CanShowGift, "Ready stable active board permits a pending gift");
        x.core.game.Fall.IsAllItemsSteady = false; Check(!teach.CanShowGift, "Falling board postpones gift"); x.core.game.Fall.IsAllItemsSteady = true;
        x.core.game.State.Busy = true; Check(!teach.CanShowGift, "Active board operations postpone gift"); x.core.game.State.Busy = false;
        x.core.game.Auto.IsThereAWaitingMatch = true; Check(!teach.CanShowGift, "Queued cascade postpones gift"); x.core.game.Auto.IsThereAWaitingMatch = false;
        x.core.WinFail.IsOutOfMoves = true; Check(!teach.CanShowGift, "Exhausted moves postpone gift before fail dialog exists"); x.core.WinFail.IsOutOfMoves = false;
        x.core.transitioning = true; Check(!teach.CanShowGift, "Scene transition postpones gift"); x.core.transitioning = false;
        x.core.failureStarted = true; Check(!teach.CanShowGift, "Failure flow postpones gift"); x.core.failureStarted = false;
        x.core.dialogs.HasActiveDialog = true; Check(!teach.CanShowGift, "Core dialog postpones gift"); x.core.dialogs.HasActiveDialog = false;

        var opening = teach.OpenGift(CancellationToken.None);
        Check(!opening.Value.IsCompleted && teach.ui.Opens == 0, "Gift waits for actual prefab prefetch");
        x.core.game.State.Playing = false; teach.ui.Prefetch.TrySetResult(true); Scheduler.Pump();
        Check(!opening.Value.IsCompleted && teach.ui.Opens == 0 && !SaveDataUtils.TeachData.newbieGiftShown,
            "Production post-prefetch gate prevents gift when board wins during load");
        Whitebox.Foundation.RoyalMatchSessionBridge.ActiveController = null; Scheduler.Pump();
        Check(!opening.Value.IsCompleted, "No core scene during handoff cannot open gift");
        Whitebox.Foundation.RoyalMatchSessionBridge.ActiveController = new(){ CoreFlow = new() };
        Scheduler.Pump(); opening.GetAwaiter().GetResult();
        Check(teach.ui.Opens == 1 && SaveDataUtils.TeachData.newbieGiftShown, "Pending gift opens once on the next stable gameplay board");
        Console.WriteLine("PASS " + passed + " effective action and presentation checks");
    }
}
