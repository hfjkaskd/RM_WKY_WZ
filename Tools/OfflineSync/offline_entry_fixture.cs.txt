public static class HttpUtil
{
    public static Action<FailHttpResponse<AccountModule.OceanShineUserReachReportResponse>> PendingGift;
    public static void RequestToServer<T>(string path, string body, Action<FailHttpResponse<T>> callback, bool block)
    {
        if (path == AccountModuleCfg.user_info)
        {
            AccountModule.Instance.InfoCalls++;
            AccountModule.Instance.PendingInfo = (Action<FailHttpResponse<AccountModule.OceanShineUserInfoResponse>>)(object)callback;
        }
        else if (path == AccountModuleCfg.new_user)
            PendingGift = (Action<FailHttpResponse<AccountModule.OceanShineUserReachReportResponse>>)(object)callback;
        else throw new Exception("Unexpected transport in isolated test: " + path);
    }
    private class Outer { public string code, Message = "fixture"; }
    public static void FailGeneric<T>(string code, Action<FailHttpResponse<T>> callback)
    {
        var outer = new Outer { code = code };
        void Reply(bool success, T data, string message, string errorCode) =>
            callback(new FailHttpResponse<T> { success = success, data = data, message = message, errorCode = errorCode });
        // HTTP_GENERIC_ERROR
    }
    public static void FailRaw(string code, Action<bool, string> callback)
    {
        var outer = new Outer { code = code };
        void Reply(bool success, string data) => callback(success, data);
        // HTTP_RAW_ERROR
    }
}
namespace Cysharp.Threading.Tasks
{
    public static class UniTask
    {
        private sealed class Pending
        {
            public Func<bool> Condition;
            public TaskCompletionSource<bool> Completion = new TaskCompletionSource<bool>();
        }
        static List<Pending> waits = new List<Pending>();
        public static int Waiting => waits.Count;
        public static Task Delay(int delay, bool ignoreTimeScale, CancellationToken cancellationToken)
        { return WaitUntil(() => true, cancellationToken, defer: true); }
        public static Task WaitUntil(Func<bool> condition, CancellationToken cancellationToken, bool defer = false)
        {
            if (!defer && condition()) return Task.CompletedTask;
            var next = new Pending { Condition = condition }; waits.Add(next); return next.Completion.Task;
        }
        public static void Advance()
        {
            var previous = waits; waits = new List<Pending>();
            foreach (var item in previous)
                if (item.Condition()) item.Completion.SetResult(true); else waits.Add(item);
        }
        public static void Reset() { waits.Clear(); }
    }
    public static class TaskExtensions
    {
        public static void Forget(this Task task)
        {
            if (task.IsFaulted) throw task.Exception;
        }
    }
}

public static class Program
{
    static int checks;
    static void Check(bool ok, string label)
    { if (!ok) throw new Exception(label); checks++; Console.WriteLine("PASS " + label); }
    static AccountModule Fresh(bool online, bool knownUser = false, bool suspended = false)
    {
        UniTask.Reset(); PlayerPrefs.Reset(); SDKAssetHandler.Dialogs = 0; Application.QuitCalls = 0; UIUtils.Tips = 0;
        Application.internetReachability = online ? NetworkReachability.ReachableViaLocalAreaNetwork : NetworkReachability.NotReachable;
        if (knownUser) PlayerPrefs.SetString(AccountModule.m_userIdKey, "known");
        if (suspended) PlayerPrefs.SetInt("WKY_AccountSuspended", 1);
        WKY_Flow.ResetCountryDecision();
        DeviceNativeBridge.NativeVpn = false;
        DeviceNativeBridge.Trace.Clear();
        UnityWebRequest.Status = 204;
        UnityWebRequest.VpnHeader = null;
        UnityWebRequest.ConfirmedPolicy = true;
        var account = new AccountModule(); account.Start().GetAwaiter().GetResult(); return account;
    }
    static void InfoOk(AccountModule account)
    {
        account.PendingInfo(new FailHttpResponse<AccountModule.OceanShineUserInfoResponse>
        { success = true, data = new AccountModule.OceanShineUserInfoResponse { Os_Uso = new AccountModule.UserInfo { Os_Cty = "BR" } } });
    }
    public static void Main()
    {
        var a = Fresh(false);
        Check(a.StartupResultAvailable && !a.AccountReady && a.LoginCalls == 0, "first offline start resolves without inventing account data");
        UniTask.Advance(); UniTask.Advance();
        Check(a.LoginCalls == 0 && UniTask.Waiting == 1, "offline retry waits for connectivity without flooding requests");
        Application.internetReachability = NetworkReachability.ReachableViaLocalAreaNetwork; UniTask.Advance();
        Check(a.LoginCalls == 1 && a.StartupResultAvailable, "reconnection sends one login and preserves the startup result");
        a.AttemptAgain().GetAwaiter().GetResult();
        Check(a.LoginCalls == 1, "in-flight attempt cannot start a second login");
        a.PendingLogin(new FailHttpResponse<AccountModule.OceanShineLoginResponse> { success = true, data = new AccountModule.OceanShineLoginResponse() });
        Check(!a.AccountReady && a.LoadingUserDataFailed, "success with missing login fields stays retryable");
        UniTask.Advance();
        a.PendingLogin(new FailHttpResponse<AccountModule.OceanShineLoginResponse>
        { success = true, data = new AccountModule.OceanShineLoginResponse { Os_Usid = "valid", Os_Aru = new AccountModule.OceanShineLoginResponse.Rule { Os_Aiu = 1 } } });
        Check(a.InfoCalls == 1 && !a.AccountReady, "valid login still waits for real user data");
        a.PendingInfo(new FailHttpResponse<AccountModule.OceanShineUserInfoResponse> { success = true, data = new AccountModule.OceanShineUserInfoResponse() });
        Check(!a.AccountReady && a.LoadingUserDataFailed, "missing nested user data cannot complete login");
        UniTask.Advance();
        int successEvents = 0; a.LoginSucceeded += () => successEvents++;
        InfoOk(a);
        Check(a.AccountReady && !a.LoadingUserDataFailed && a.LoadingUserDataErrorCode == null && successEvents == 1,
            "recovered profile clears failure and publishes one success event");
        int calls = a.InfoCalls; UniTask.Advance(); a.AttemptAgain().GetAwaiter().GetResult();
        Check(a.InfoCalls == calls, "success stops background retries");

        a = Fresh(false, true);
        Check(a.StartupResultAvailable && a.LoginCalls == 0, "returning user can enter offline");
        a = Fresh(false, true, true);
        Check(!a.StartupResultAvailable && !a.IsSuspended && a.HasLocalSuspension && UIUtils.Tips == 1,
            "local suspension requires an online decision and reports the connection problem");
        Application.internetReachability = NetworkReachability.ReachableViaLocalAreaNetwork; UniTask.Advance();
        a.PendingInfo(new FailHttpResponse<AccountModule.OceanShineUserInfoResponse> { success = false, errorCode = "NETWORK_TIMEOUT" });
        Check(!a.StartupResultAvailable && UIUtils.Tips == 1, "timeout neither clears a saved suspension nor repeats its first-failure notice");
        UniTask.Advance(); InfoOk(a);
        Check(a.AccountReady && a.StartupResultAvailable && !a.HasLocalSuspension, "valid server profile clears saved suspension");
        a = Fresh(true);
        a.PendingLogin(new FailHttpResponse<AccountModule.OceanShineLoginResponse> { success = true, errorCode = "1001" });
        Check(a.IsSuspended && a.HasLocalSuspension && SDKAssetHandler.Dialogs == 1, "server suspension persists and shows one dialog");
        a.PendingLogin(new FailHttpResponse<AccountModule.OceanShineLoginResponse> { success = true, errorCode = "1001" });
        UniTask.Advance(); a.AttemptAgain().GetAwaiter().GetResult();
        Check(SDKAssetHandler.Dialogs == 1 && a.LoginCalls == 1, "suspension stops retries and duplicate dialogs");
        Check(Application.QuitCalls == 0 && SDKAssetHandler.LastDialog.des == "HTTPNetworkProblem",
            "source suspension dialog does not quit before it is dismissed");
        SDKAssetHandler.LastDialog.onFinish();
        Check(Application.QuitCalls == 1, "dismissing the source suspension dialog quits the application");

        a = Fresh(true, true); InfoOk(a);
        int ordinaryCallbacks = 0;
        HttpUtil.FailGeneric<object>("1001", response =>
        {
            ordinaryCallbacks++;
            Check(response.success && response.data == null && response.errorCode == "1001",
                "ordinary typed business error keeps its existing callback contract");
        });
        Check(ordinaryCallbacks == 1 && a.AccountReady && !a.IsSuspended && !a.HasLocalSuspension && SDKAssetHandler.Dialogs == 0,
            "ordinary typed requests cannot suspend or force an in-game exit");
        HttpUtil.FailRaw("1001", (success, data) =>
        {
            ordinaryCallbacks++;
            Check(!success && data == null, "ordinary raw business error keeps its existing callback contract");
        });
        Check(ordinaryCallbacks == 2 && a.AccountReady && !a.IsSuspended && !a.HasLocalSuspension && SDKAssetHandler.Dialogs == 0,
            "ordinary raw requests cannot suspend or force an in-game exit");

        bool profileCallbackSawSuspension = false;
        a.Request_UserInfoRequest(true, false, response => profileCallbackSawSuspension = a.IsSuspended);
        HttpUtil.FailGeneric("1001", a.PendingInfo);
        Check(profileCallbackSawSuspension && a.IsSuspended && a.HasLocalSuspension && SDKAssetHandler.Dialogs == 1,
            "in-game profile refresh detects suspension before its caller callback, matching source");

        a = Fresh(true, true);
        HttpUtil.FailGeneric("1001", a.PendingInfo);
        Check(a.IsSuspended && !a.AccountReady && SDKAssetHandler.Dialogs == 1,
            "startup profile request preserves the source suspension gate");

        a = Fresh(true, true); InfoOk(a);
        bool giftCallbackSawSuspension = false;
        a.Request_UserReachReportRequest(response => giftCallbackSawSuspension = a.IsSuspended);
        HttpUtil.FailGeneric("1001", HttpUtil.PendingGift);
        Check(giftCallbackSawSuspension && a.IsSuspended && a.HasLocalSuspension && SDKAssetHandler.Dialogs == 1,
            "newbie reward detects suspension before its caller callback");

        a = Fresh(true, true); InfoOk(a);
        a.Request_UserInfoRequest(true, false);
        HttpUtil.FailGeneric("500", a.PendingInfo);
        Check(!a.IsSuspended && SDKAssetHandler.Dialogs == 0 && a.AccountReady,
            "other profile business errors never become a suspension");

        a = Fresh(false); OverseaPlatform.AdSdk = null;
        int failed = 0, finished = 0;
        var ad = new ShowAdArgs { onFailed = () => failed++, onFinish = result => { Check(!result.success, "offline ad reports failure"); finished++; } };
        BizzaSdk.Ad.ShowRewardAd(ad); BizzaSdk.Ad.ShowInterAd(ad);
        Check(failed == 2 && finished == 2, "both ad entry points complete before SDK exists");
        a = Fresh(true, true); InfoOk(a); OverseaPlatform.AdSdk = new AggregationAdSdk();
        BizzaSdk.Ad.ShowRewardAd(new ShowAdArgs()); BizzaSdk.Ad.ShowInterAd();
        Check(OverseaPlatform.AdSdk.RewardCalls == 1 && OverseaPlatform.AdSdk.InterCalls == 1, "ready account preserves real ad calls");
        a.PendingInfo(new FailHttpResponse<AccountModule.OceanShineUserInfoResponse> { success = true, errorCode = "1001" });
        BizzaSdk.Ad.ShowRewardAd(ad);
        Check(OverseaPlatform.AdSdk.RewardCalls == 1 && finished == 3, "suspension after entry blocks ads too");

        a = Fresh(false);
        UnityWebRequest.Status = 0;
        Application.internetReachability = NetworkReachability.ReachableViaLocalAreaNetwork;
        UniTask.Advance();
        Check(a.StartupResultAvailable && !a.AccountReady && a.LoginCalls == 0,
            "unconfirmed region permits only basic gameplay without login traffic");
        UnityWebRequest.Status = 204; UniTask.Advance();
        Check(a.LoginCalls == 1, "confirmed region permits one resumed login");

        a = Fresh(true, true); InfoOk(a); OverseaPlatform.AdSdk = new AggregationAdSdk();
        int adFinishes = 0, adFailures = 0;
        BizzaSdk.Ad.ShowRewardAd(new ShowAdArgs { onFailed = () => adFailures++, onFinish = result =>
        {
            Check(!result.success, "failed adapter cannot award a reward");
            adFinishes++;
        } });
        OverseaPlatform.AdSdk.LastArgs.onFailed();
        OverseaPlatform.AdSdk.LastArgs.onFinish(new ShowAdResult { success = true });
        Check(adFinishes == 1 && adFailures == 1, "duplicate adapter callbacks complete the caller once");
        Check(UIUtils.Tips == 1, "ready account with unavailable ad receives one network notice");

        foreach (string setting in new[] { "false", "0", "true", null, "invalid" })
        {
            a = Fresh(false);
            DeviceNativeBridge.NativeVpn = true;
            UnityWebRequest.VpnHeader = setting;
            Application.internetReachability = NetworkReachability.ReachableViaLocalAreaNetwork;
            UniTask.Advance();
            bool allowed = setting == "false" || setting == "0";
            Check(DeviceNativeBridge.Trace.Count == 2 && DeviceNativeBridge.Trace[0] == "Worker" &&
                DeviceNativeBridge.Trace[1] == "VPN", "Worker decision precedes native VPN: " + (setting ?? "missing"));
            Check((a.LoginCalls == 1) == allowed && (SDKAssetHandler.Dialogs == 0) == allowed &&
                DeviceInfoUtil.LastVpn != allowed && DeviceNativeBridge.IsVpnDetectionEnabled != allowed,
                "VPN decision uses confirmed switch: " + (setting ?? "missing"));
            if (!allowed)
            {
                UniTask.Advance();
                Check(SDKAssetHandler.Dialogs == 1 && SDKAssetHandler.LastDialog.des == "TokenClient_VPN" &&
                    !WKY_Flow.IsCountryAccessAllowed && UniTask.Waiting == 0,
                    "VPN denial stops account retries");
            }
        }

        a = Fresh(false);
        UnityWebRequest.Status = 403;
        Application.internetReachability = NetworkReachability.ReachableViaLocalAreaNetwork;
        UniTask.Advance();
        Check(a.LoginCalls == 0 && WKY_Flow.IsAccessDenied && PlayerPrefs.GetInt("WKY_CountryDenied", 0) == 1 &&
            SDKAssetHandler.LastDialog.des == "TokenClient_REGION" && !DeviceNativeBridge.Trace.Contains("VPN"),
            "confirmed country denial takes precedence and is persisted");
        UnityWebRequest.Status = 0;
        Check(!WKY_Flow.ProbeCountryPolicy().GetAwaiter().GetResult() && PlayerPrefs.GetInt("WKY_CountryDenied", 0) == 1,
            "Worker unavailability never clears confirmed denial");
        UnityWebRequest.Status = 204;
        Check(WKY_Flow.ProbeCountryPolicy().GetAwaiter().GetResult() && !PlayerPrefs.HasKey("WKY_CountryDenied"),
            "confirmed Worker allow clears the previous country denial");

        a = Fresh(false);
        DeviceNativeBridge.NativeVpn = true;
        UnityWebRequest.Status = 0;
        UnityWebRequest.VpnHeader = "false";
        Application.internetReachability = NetworkReachability.ReachableViaLocalAreaNetwork;
        UniTask.Advance();
        Check(a.LoginCalls == 0 && WKY_Flow.IsAccessDenied && DeviceNativeBridge.IsVpnDetectionEnabled &&
            SDKAssetHandler.LastDialog.des == "TokenClient_VPN", "unavailable Worker cannot apply an unconfirmed VPN toggle");

        Console.WriteLine("OFFLINE_ENTRY_PASS cases=" + checks);
    }
