[AsyncMethodBuilder(typeof(AsyncUniTaskMethodBuilder))]
struct UniTask {
    public Task Value;
    public UniTask(Task value) { Value=value; }
    public TaskAwaiter GetAwaiter() => (Value ?? Task.CompletedTask).GetAwaiter();

    public static UniTask NextFrame(CancellationToken token=default) => Scheduler.Frame(token);

static class Scheduler {
    static readonly List<(Func<bool> predicate,CancellationToken token,TaskCompletionSource<bool> completion)> waits=new();
    static readonly List<(CancellationToken token,TaskCompletionSource<bool> completion)> frames=new();
    public static UniTask Wait(Func<bool> predicate,CancellationToken token) {
        token.ThrowIfCancellationRequested(); if(predicate()) return new(Task.CompletedTask);
        var completion=new TaskCompletionSource<bool>(); waits.Add((predicate,token,completion)); return new(completion.Task);
    }
    public static UniTask Frame(CancellationToken token) {
        token.ThrowIfCancellationRequested(); var completion=new TaskCompletionSource<bool>();frames.Add((token,completion));return new(completion.Task);
    }
    public static void Pump() {
        var ready=frames.ToArray();frames.Clear();
        foreach(var item in ready) { if(item.token.IsCancellationRequested)item.completion.TrySetCanceled();else item.completion.TrySetResult(true); }
        foreach(var item in waits.ToArray()) {
            if(!item.token.IsCancellationRequested && !item.predicate())continue;
            waits.Remove(item);if(item.token.IsCancellationRequested)item.completion.TrySetCanceled();else item.completion.TrySetResult(true);
        }
        Time.frameCount++;
    }
    public static void Finish(UniTask task) {
        for(int i=0;!task.Value.IsCompleted && i<200;i++){Pump();Thread.Sleep(1);}
        if(!task.Value.IsCompleted)throw new Exception("Unresolved async boundary");task.GetAwaiter().GetResult();
    }
}
