class Runner {
    static int checks;
    static void Check(bool ok, string message) { if(!ok) throw new Exception(message); checks++; Console.WriteLine("PASS " + message); }
    static void Reset() {
        AccountModule.Instance=new(); AccountModule.CountryType=AccountModule.E_CountryType.None;
        SaveDataUtils.GameData=new(); SaveDataUtils.TeachData=new(); SaveDataUtils.Saves=0;
    }
    static void Main() {
        Reset(); var module=new Teach01Module(); module.Start();
        Check(module.Runs==1,"offline first-time player starts the gift flow without a server profile");
        module.OnFirstMatch(); module.OnFirstMatch();
        Check(SaveDataUtils.TeachData.newbieGiftRequest==NewbieGiftRequestState.Background && SaveDataUtils.Saves==1,
            "first offline match persists one background claim; later matches cannot repeat it");
        Reset(); AccountModule.Instance.AccountReady=true; AccountModule.Instance.Os_Current_Uso=new(){Os_Ncm=true};
        module=new(); module.Start(); module.OnFirstMatch();
        Check(SaveDataUtils.TeachData.newbieGiftRequest==NewbieGiftRequestState.NextLogin,"first online match persists the next-login retry state before HTTP");
        foreach(var state in new[]{NewbieGiftRequestState.Background,NewbieGiftRequestState.NextLogin}) {
            Reset(); SaveDataUtils.TeachData.newbieGiftRequest=state; SaveDataUtils.TeachData.completed=true;
            SaveDataUtils.GameData.customTutorialEnd=true; AccountModule.Instance.Os_Current_Uso=new(){Os_Ncm=false};
            module=new(); module.Start(); module.Start();
            Check(module.Runs==1,"restored pending claim bypasses completed teaching and old-user flag: "+state);
        }
        Reset(); SaveDataUtils.TeachData.completed=true; module=new(); module.Start();
        Check(module.Runs==0,"legacy completed user is not granted a new gift");
        Reset(); SaveDataUtils.TeachData.newbieGiftRequest=NewbieGiftRequestState.Succeeded; SaveDataUtils.TeachData.newbieGiftShown=true;
        module=new(); module.Start(); Check(module.Runs==0,"confirmed and shown gift never starts again");
        Reset(); AccountModule.Instance.IsSuspended=true; module=new(); module.Start();
        Check(module.Runs==0,"suspended account cannot start gift requests");
        foreach(bool single in new[]{false,true}) {
            Reset(); ChannelConfig.Instance.real_CustomConfig.singleCurrencyMode=single;
            SaveDataUtils.TeachData.newbieGiftRequest=NewbieGiftRequestState.Background;
            ItemUtils.Grants=0; var page=new GiftPage(); page.Open(); page.OnButtonClick();
            Check(page.Os_ComText.text==(single?"R$0,40":"R$51,00"),"offline Brazilian gift amount: "+single);
            Check(page.Closed && ItemUtils.Grants==0,"offline close never grants local currency: "+single);
            page.OnButtonClick();
            Check(ItemUtils.Grants==0 && AccountModule.CountryType==AccountModule.E_CountryType.None,
                "repeated offline close neither grants currency nor changes global country: "+single);
        }
        Reset(); AccountModule.CountryType=AccountModule.E_CountryType.BR; ItemUtils.Grants=0;
        var online=new GiftPage(); online.Open(); online.OnButtonClick();
        Check(online.Closed && ItemUtils.Grants==1,"normal online gift preserves its existing award action");
        online.OnButtonClick();
        Check(ItemUtils.Grants==1,"repeated click during close cannot grant the normal gift twice");
        var words=new Words(); words.Dict["en-US"]="English fallback"; TableUtils.Tables.TblLanguage.DataMap["sample"]=words;
        Check(LanguageUtils.GetText("sample")=="English fallback","missing translation falls back to English");
        words.Dict["pt-BR"]=""; Check(LanguageUtils.GetText("sample")=="English fallback","empty translation also falls back to English");
        words.Dict["pt-BR"]="Portuguese"; Check(LanguageUtils.GetText("sample")=="Portuguese","available translation remains selected");
        Check(LanguageUtils.GetText("missing")=="missing","missing language key is handled safely");
        Reset(); SDKAssetHandler.Opens=0;
        Check(!UIUtils.CheckWKYAccountReady() && SDKAssetHandler.Opens==1 && SDKAssetHandler.Last.des=="WKY_Reconnecting" && SDKAssetHandler.Last.isLanguage,
            "unready business action opens the localized reconnect dialog");
        AccountModule.Instance.AccountReady=true; Check(UIUtils.CheckWKYAccountReady() && SDKAssetHandler.Opens==1,"ready business action opens no reconnect dialog");
        AccountModule.Instance.IsSuspended=true; Check(!UIUtils.CheckWKYAccountReady() && SDKAssetHandler.Opens==1,"suspension does not add a second reconnect dialog");
        Console.WriteLine("OFFLINE_GIFT_PASS cases="+checks);
    }
}
